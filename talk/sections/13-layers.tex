    % column 1: kernel
    \node[chead, text=PennGreen] at (0.15,-0.2) {1. Validity};
    \node[ctag] at (0.15,-0.62) {The Collatz ``disproof'', Jul 2026};
    \node[cbody] at (0.15,-1.05) {%
      \begin{itemize}\setlength{\itemsep}{3pt}
        \item A Lean repository with no \texttt{sorry} ``disproved'' the Collatz conjecture.
        \item It passed Lean's kernel and an independent checker. It was built on purpose, from a kernel bug that allowed a proof of \emph{False}.
        \item Fixed about an hour after it was reduced to a proof of \emph{False}. The advisory covers every Lean~4 version up to v4.31.
      \end{itemize}};
    % column 2: statement
    \node[chead, text=PennGold!85!black] at (4.95,-0.2) {2. Statement fidelity};
    \node[ctag] at (4.95,-0.62) {Proved, but not what was meant};
    \node[cbody] at (4.95,-1.05) {%
      \begin{itemize}\setlength{\itemsep}{3pt}
        \item An OEIS conjecture ``disproved'' by AI for \$0.17 of API spend. The formal statement was wrong.
        \item Erd\H{o}s problem \#124 was restated after AI attempts showed it was unclear which density was meant.
        \item Lean's conventions can change a meaning: the infimum of an empty set of naturals is $0$.
      \end{itemize}};
    % column 3: proof
    \node[chead, text=PennRed] at (9.75,-0.2) {3. Proof fidelity};
    \node[ctag] at (9.75,-0.62) {Correct, but not the written argument};
    \node[cbody] at (9.75,-1.05) {%
      \begin{itemize}\setlength{\itemsep}{3pt}
        \item Navier--Stokes: the paper's estimate needs $m{+}4$ derivatives; the Lean lemma assumes $m{+}5$. A pressure bound depends on an extra quantity and takes another route.
        \item Fermat's Last Theorem: about 13M lines, ``likely far longer than necessary''.
      \end{itemize}};

  \end{tikzpicture}
